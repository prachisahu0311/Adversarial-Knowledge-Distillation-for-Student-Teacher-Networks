# Define function to count parameters
def count_parameters(model):
    trainable = sum(p.numel() for p in model.parameters() if p.requires_grad)
    non_trainable = sum(p.numel() for p in model.parameters() if not p.requires_grad)
    return trainable, non_trainable

# Define Teacher Model (ResNet-34)

# Define Student20 Model (Simplified Custom CNN, ~20% of teacher parameters)
class Student20(nn.Module):
    def __init__(self):
        super(Student20, self).__init__()
        self.conv1 = nn.Conv2d(3, 6, 3, padding=1)
        self.bn1 = nn.BatchNorm2d(6)
        self.conv2 = nn.Conv2d(6, 12, 3, padding=1)
        self.bn2 = nn.BatchNorm2d(12)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc1 = nn.Linear(12 * 56 * 56, 100)
        self.dropout = nn.Dropout(0.5)
        self.fc2 = nn.Linear(100, 100)

    def forward(self, x):
        x = self.pool(F.relu(self.bn1(self.conv1(x))))
        x = self.pool(F.relu(self.bn2(self.conv2(x))))
        x = x.view(-1, 12 * 56 * 56)
        x = F.relu(self.fc1(x))
        x = self.dropout(x)
        x = self.fc2(x)
        return x

# Define Student50 Model (Modified ResNet-18, ~50% of teacher parameters)
class Student50(nn.Module):
    def __init__(self):
        super(Student50, self).__init__()
        base = torchvision.models.resnet18(weights=None)
        base.conv1 = nn.Conv2d(16, 64, kernel_size=7, stride=2, padding=3, bias=False)
        self.features = nn.Sequential(*list(base.children())[:-2])  # Remove last two layers
        self.conv1 = nn.Conv2d(3, 16, 3, stride=1, padding=1)  # Custom first layer
        self.bn1 = nn.BatchNorm2d(16)
        self.pool = nn.AdaptiveAvgPool2d((1, 1))
        self.fc = nn.Linear(512, 100)  # CIFAR-100 classes

    def forward(self, x):
        x = F.relu(self.bn1(self.conv1(x)))
        x = self.features(x)
        x = self.pool(x)
        x = x.view(x.size(0), -1)
        x = self.fc(x)
        return x

# Define Generator Network (optimized for faster computation)
class Generator(nn.Module):
    def __init__(self, z_dim=100):
        super(Generator, self).__init__()
        self.model = nn.Sequential(
            nn.ConvTranspose2d(z_dim, 128, 4, 1, 0, bias=False),
            nn.BatchNorm2d(128),
            nn.ReLU(True),
            nn.ConvTranspose2d(128, 64, 4, 2, 1, bias=False),
            nn.BatchNorm2d(64),
            nn.ReLU(True),
            nn.ConvTranspose2d(64, 32, 4, 2, 1, bias=False),
            nn.BatchNorm2d(32),
            nn.ReLU(True),
            nn.ConvTranspose2d(32, 3, 4, 2, 1, bias=False),
            nn.Tanh()
        )

    def forward(self, x):
        return self.model(x)

# Loss functions
criterion_kl = nn.KLDivLoss(reduction='batchmean')

# Optimizers
optimizer_g = optim.Adam(generator.parameters(), lr=0.0002, betas=(0.5, 0.999))
optimizer_s20 = optim.Adam(student20.parameters(), lr=0.001)

# Training function for AKD
def train_akd(student, optimizer_s, testloader, split_name, epochs=150, model_name="Student"):
    z_dim = 100
    batch_size = 64
    accum_steps = 2  # Gradient accumulation steps

    for epoch in range(epochs):
        student.train()
        generator.train()
        running_loss_s = 0.0
        running_loss_g = 0.0
        optimizer_s.zero_grad()
        optimizer_g.zero_grad()

        for i in range(50):  # Simulate 50 batches per epoch
            z = torch.randn(batch_size, z_dim, 1, 1).to(device)
            fake_images = generator(z)  # 32x32 images
            fake_images_upscaled = F.interpolate(fake_images, size=(224, 224), mode='bilinear', align_corners=False)

            # Get teacher and student outputs
            with torch.no_grad():
                teacher_logits = teacher_model(fake_images_upscaled)
            student_logits = student(fake_images_upscaled)

            # Compute student loss
            loss_kl = criterion_kl(
                F.log_softmax(student_logits, dim=1),
                F.softmax(teacher_logits, dim=1)
            ) / accum_steps

            # Update student with gradient accumulation
            loss_kl.backward()
            if (i + 1) % accum_steps == 0:
                optimizer_s.step()
                optimizer_s.zero_grad()

            # Compute generator loss
            fake_images = generator(z)  # Regenerate to avoid graph issues
            fake_images_upscaled = F.interpolate(fake_images, size=(224, 224), mode='bilinear', align_corners=False)
            with torch.no_grad():
                teacher_logits = teacher_model(fake_images_upscaled)
            student_logits = student(fake_images_upscaled)
            loss_g = -criterion_kl(
                F.log_softmax(student_logits, dim=1),
                F.softmax(teacher_logits, dim=1)
            ) / accum_steps

            # Update generator with gradient accumulation
            loss_g.backward()
            if (i + 1) % accum_steps == 0:
                optimizer_g.step()
                optimizer_g.zero_grad()

            running_loss_s += loss_kl.item() * accum_steps
            running_loss_g += loss_g.item() * accum_steps

        if (epoch + 1) % 50 == 0:
            print(f"[{model_name}] Epoch {epoch+1}/{epochs}, Student Loss: {running_loss_s/50:.4f}, Generator Loss: {running_loss_g/50:.4f}")
            # Evaluate test accuracy every 50 epochs on smaller subset
            evaluate_model(student, testloader, model_name, split_name, epoch + 1)

        if (epoch + 1) % 100 == 0:
            with torch.no_grad():
                fake_images = generator(torch.randn(9, z_dim, 1, 1).to(device))
                fake_images = fake_images.cpu().detach()
                plt.figure(figsize=(6, 6))
                for i in range(9):
                    plt.subplot(3, 3, i+1)
                    plt.imshow(fake_images[i].permute(1, 2, 0).numpy() * 0.5 + 0.5)
                    plt.axis('off')
                plt.savefig(f"generated_images_{model_name}_epoch_{epoch+1}.png")
                plt.close()

# Evaluation function
def evaluate_model(model, testloader, model_name, split_name, epoch):
    model.eval()
    correct = 0
    total = 0
    all_preds = []
    all_labels = []

    with torch.no_grad():
        for images, labels in testloader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            _, predicted = torch.max(outputs.data, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
            all_preds.extend(predicted.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())

    accuracy = 100 * correct / total
    print(f"{model_name} Accuracy on {split_name} (Epoch {epoch}): {accuracy:.2f}%")

    # Confusion Matrix (only at final evaluation)
    if epoch == 150:
        cm = confusion_matrix(all_labels, all_preds)
        plt.figure(figsize=(10, 8))
        sns.heatmap(cm, annot=False, cmap='Blues')
        plt.title(f"Confusion Matrix - {model_name} ({split_name})")
        plt.xlabel('Predicted')
        plt.ylabel('True')
        plt.savefig(f"cm_{model_name}_{split_name}.png")
        plt.close()

    return accuracy
